def qty_allowed(qty, stock):
    if type(qty) is not int or type(stock) is not int:
        return False
    if qty < 1:
        return False
    return qty <= stock

assert qty_allowed(1, 10)
assert qty_allowed(10, 10)
assert not qty_allowed(11, 10)
assert not qty_allowed("11", 10)
